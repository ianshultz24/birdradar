function tierB(tier, t) {
  if (tier === 'lifer' || tier === 'lifer-rare') return { color: t.lifer, bg: t.liferBg, border: t.liferBorder };
  if (tier === 'rare') return { color: t.rare, bg: t.rareBg, border: t.rareBorder };
  return { color: t.seen, bg: t.seenBg, border: t.seenBorder };
}

function tierLabelB(tier) {
  if (tier === 'lifer-rare') return 'Lifer · Rare';
  if (tier === 'lifer') return 'Lifer';
  if (tier === 'rare') return 'Rare';
  return 'Seen';
}

function AlertsPanelB({ t, mono, sans, display, obs, targets, focus, toggleFocus, sort, setSort, sorted, lifers, rare, seen }) {
  const [query, setQuery] = useStateB('');
  const q = query.toLowerCase().trim();
  const searching = !!q;
  const results = searching ? obs.filter(o => o.comName.toLowerCase().includes(q) || o.sciName.toLowerCase().includes(q)) : [];

  return (
    <div>
      <div style={{ padding: '14px 20px', borderBottom: `1px solid ${t.line1}` }}>
        <div style={{ position: 'relative' }}>
          <Icons.Search size={15} style={{ position: 'absolute', left: 10, top: '50%', transform: 'translateY(-50%)', color: t.fg3 }}/>
          <input value={query} onChange={e => setQuery(e.target.value)} placeholder="Search species nearby…"
            style={{ width: '100%', padding: '9px 34px 9px 34px', fontSize: 13, fontFamily: sans,
              background: t.bg2, border: `1px solid ${searching ? t.accentBorder : t.line2}`,
              borderRadius: 8, outline: 'none', color: t.fg0, transition: 'border-color 0.15s' }}/>
          {searching && (
            <button onClick={() => setQuery('')} style={{ position: 'absolute', right: 10, top: '50%', transform: 'translateY(-50%)', background: 'transparent', border: 'none', color: t.fg3, cursor: 'pointer' }}>
              <Icons.X size={14}/>
            </button>
          )}
        </div>
        {!searching && (
          <div style={{ display: 'flex', marginTop: 10, border: `1px solid ${t.line2}`, borderRadius: 8, overflow: 'hidden' }}>
            {['recent','closest'].map(s => (
              <button key={s} onClick={() => setSort(s)} style={{
                flex: 1, padding: '7px 0',
                background: sort === s ? t.accentBg : 'transparent',
                border: 'none',
                color: sort === s ? t.accent : t.fg2,
                fontSize: 12, fontWeight: sort === s ? 600 : 400, cursor: 'pointer', fontFamily: sans,
              }}>{s === 'recent' ? 'Most Recent' : 'Closest'}</button>
            ))}
          </div>
        )}
      </div>

      {searching ? (
        <>
          <div style={{ padding: '10px 20px', fontSize: 12, color: t.fg3, borderBottom: `1px solid ${t.line1}`, fontFamily: mono }}>
            {results.length} result{results.length !== 1 ? 's' : ''}
          </div>
          {results.map(o => <ObsCardB key={`s-${o.speciesCode}-${o.locName}`} obs={o} t={t} mono={mono} sans={sans} display={display} focus={focus} toggleFocus={toggleFocus}/>)}
        </>
      ) : (
        <>
          {targets.length > 0 && (
            <>
              <SectionHeaderB title="Target Species" count={targets.length} color={t.target} t={t} mono={mono}/>
              {targets.map(sp => <TargetCardB key={sp.speciesCode} sp={sp} t={t} mono={mono} display={display}/>)}
            </>
          )}
          <SectionHeaderB title="Lifer Opportunities" count={lifers.length} color={t.lifer} t={t} mono={mono}/>
          {sorted(lifers).map(o => <ObsCardB key={`${o.speciesCode}-${o.locName}`} obs={o} t={t} mono={mono} sans={sans} display={display} focus={focus} toggleFocus={toggleFocus}/>)}
          <SectionHeaderB title="Rare — Already Seen" count={rare.length} color={t.rare} t={t} mono={mono}/>
          {sorted(rare).map(o => <ObsCardB key={`${o.speciesCode}-${o.locName}`} obs={o} t={t} mono={mono} sans={sans} display={display} focus={focus} toggleFocus={toggleFocus}/>)}
          <SectionHeaderB title="Seen Nearby" count={seen.length} color={t.seen} t={t} mono={mono}/>
          {sorted(seen).map(o => <ObsCardB key={`${o.speciesCode}-${o.locName}`} obs={o} t={t} mono={mono} sans={sans} display={display} focus={focus} toggleFocus={toggleFocus}/>)}
        </>
      )}
    </div>
  );
}

                <div style={{ fontSize: 13, fontWeight: 600, fontFamily: display, color: t.fg0 }}>{obs.comName}</div>

                  {fmtDist(obs.distKm)} {obs.bearingLabel} · {fmtTimeAgo(obs.obsDt)}
                </div>
              </div>
            )}
          </div>
        );
      })}

      {/* User dot */}
      <div style={{ position: 'absolute', left: '50%', top: '50%', transform: 'translate(-50%, -50%)', zIndex: 4 }}>
        <div style={{ width: 12, height: 12, borderRadius: '50%', background: t.accent, border: `2.5px solid ${t.bg1}`, boxShadow: t.shadow }}/>
      </div>

      {/* Map controls */}
      <div style={{ position: 'absolute', top: 12, right: 12, display: 'flex', flexDirection: 'column', borderRadius: 8, overflow: 'hidden', boxShadow: t.shadow, border: `1px solid ${t.line2}` }}>
        {['+', '−'].map((s, i) => (
          <button key={s} style={{ width: 36, height: 36, background: t.cardBg, border: 'none', borderBottom: i === 0 ? `1px solid ${t.line1}` : 'none',
            color: t.fg1, fontSize: 16, cursor: 'pointer' }}>{s}</button>
        ))}
      </div>

      {/* Status pills */}
      <div style={{ position: 'absolute', top: 12, left: 12, display: 'flex', gap: 6 }}>
        {[
          { label: 'Lifers', value: OBS.filter(o => o.tier==='lifer'||o.tier==='lifer-rare').length, color: t.lifer },
          { label: 'Rare', value: OBS.filter(o => o.tier==='rare').length, color: t.rare },
          { label: 'Total', value: OBS.length },
        ].map(p => (
          <div key={p.label} style={{ padding: '5px 10px', background: t.cardBg, border: `1px solid ${t.line2}`, borderRadius: 6,
            display: 'flex', alignItems: 'center', gap: 6, boxShadow: t.shadow, fontFamily: mono, fontSize: 11 }}>
            <span style={{ color: t.fg3 }}>{p.label}</span>
            <span style={{ fontWeight: 600, color: p.color || t.fg0 }}>{p.value}</span>
          </div>
        ))}
      </div>

      {/* Coords */}
      <div style={{ position: 'absolute', bottom: 12, right: 12, padding: '5px 10px', background: t.cardBg, border: `1px solid ${t.line2}`,
        borderRadius: 6, fontFamily: mono, fontSize: 10, color: t.fg2, boxShadow: t.shadow }}>
        47.6526°N, 122.1714°W
      </div>

      <div style={{ position: 'absolute', bottom: 12, left: 12 }}>
        <button style={{ padding: '7px 14px', display: 'flex', alignItems: 'center', gap: 6, background: t.cardBg,
          border: `1px solid ${t.line2}`, borderRadius: 8, color: t.fg1, fontSize: 12, fontWeight: 500, cursor: 'pointer',
          fontFamily: sans, boxShadow: t.shadow }}>
          <Icons.RefreshCw size={13}/> Refresh
        </button>
      </div>
    </div>
  );
}
